# pipeline/entity_linker.py
class EntityLinker:
    """实体链接器"""
    
    def __init__(self, char_manager=None):
        self.char_manager = char_manager or get_character_manager()
        self._link_cache = {}
    
    def link(self, entities: List, text: str, char_manager: CharacterManager = None) -> List[LinkedEntity]:
        """链接实体到角色库"""
        # 修复：使用局部变量而非修改实例属性
        mgr = char_manager or self.char_manager
        
        linked_results = []
        for entity in entities:
            if entity.type != 'PER':
                linked_results.append(LinkedEntity(
                    text=entity.text,
                    type=entity.type,
                    confidence=entity.confidence,
                    start_pos=entity.start,
                    end_pos=entity.end
                ))
                continue
            
            # 使用局部变量 mgr
            linked_name = self._try_link(entity.text, mgr)
            if linked_name:
                linked_results.append(LinkedEntity(
                    text=entity.text,
                    type=entity.type,
                    confidence=max(entity.confidence, 0.8),
                    standard_name=linked_name,
                    is_linked=True,
                    start_pos=entity.start,
                    end_pos=entity.end
                ))
            else:
                linked_results.append(LinkedEntity(
                    text=entity.text,
                    type=entity.type,
                    confidence=entity.confidence,
                    start_pos=entity.start,
                    end_pos=entity.end
                ))
        
        return linked_results
    
    def _try_link(self, entity_text: str, char_manager: CharacterManager) -> Optional[str]:
        """尝试将实体文本链接到角色库标准名"""
        # 直接匹配
        char = char_manager.get_character_by_name(entity_text)
        if char:
            return char.name
        
        # 别名匹配
        char = char_manager.get_character_by_alias(entity_text)
        if char:
            return char.name
        
        # 核心名提取（如"张总" → "张"）
        if len(entity_text) >= 2 and entity_text[-1] in '总哥姐叔伯':
            core_name = entity_text[:-1]
            char = char_manager.get_character_by_name(core_name)
            if char:
                return char.name
        
        return None

class SpeakerRoleFilter:
    """说话角色过滤器"""
    
    def __init__(self, semantic_ranker=None, l2_threshold=0.7):
        self.semantic_ranker = semantic_ranker
        self.l2_threshold = l2_threshold
        self._dialogue_entities = set()
        self._entity_scene_count = {}
    
    def filter(self, entities: List, text: str, nlp) -> List:
        """过滤非说话角色的PER实体"""
        # 修复：每次调用都重置状态
        self._dialogue_entities.clear()
        self._entity_scene_count.clear()
        
        # 提取当前文本的对话上下文
        self._extract_dialogue_entities(text, nlp)
        
        # 过滤逻辑
        filtered = []
        for entity in entities:
            if entity.type != 'PER':
                # ORG/LOC 直接放行
                filtered.append(entity)
                continue
            
            if entity.text in self._dialogue_entities:
                # 说话角色，保留
                filtered.append(entity)
            else:
                # 非说话角色，检查是否需要L2增强
                if self._should_boost(entity):
                    entity.confidence = self._apply_l2_boost(entity)
                    filtered.append(entity)
        
        return filtered

### T-06: 修复引号内句子拆分逻辑

**问题**: H-04 - 引号内句子拆分错误
**影响**: 直接影响说话人匹配准确率

#### 实施步骤

**步骤1**: 在 `utils/` 目录下创建 `text_utils.py`

```python
# utils/text_utils.py
# -*- coding: utf-8 -*-
"""文本处理工具"""
import re
from typing import List


def split_sentences_smart(text: str) -> List[str]:
    """
    智能句子分割，考虑引号完整性
    
    示例:
        输入: '"你好吗？他问。"苏夜回答道。'
        输出: ['"你好吗？他问。"', '苏夜回答道。']
    """
    if not text:
        return []
    
    # 策略：使用正则匹配引号内的内容和引号外的内容
    # 匹配模式：引号内容 或 非引号内容（以。！？结尾）
    pattern = r'(["\u201c][^"\u201d]*["\u201d][。！？]?|[^"\u201c。！？]*[。！？])'
    
    sentences = re.findall(pattern, text)
    
    # 过滤空字符串
    return [s.strip() for s in sentences if s.strip()]


def split_sentences_simple(text: str) -> List[str]:
    """
    简单句子分割（不考虑引号）
    
    适用于已知文本不含引号的场景
    """
    if not text:
        return []
    
    sentences = re.split(r'[。！？]', text)
    return [s.strip() for s in sentences if s.strip()]
```

# pipeline/entity_clusterer.py
class EntityClusterer:
    """实体聚类器"""
    
    def cluster(self, char_manager: CharacterManager = None, write_back: bool = False) -> Dict[str, List[str]]:
        """
        执行实体聚类
        
        返回:
            聚类结果映射字典 {原实体名: 中心实体名}
        """
        mgr = char_manager or self.char_manager
        
        # 步骤1-3: 聚类逻辑
        entity_contexts = self._collect_contexts(mgr)
        entity_vectors = self._encode_vectors(entity_contexts)
        clusters = self._cluster_entities(entity_vectors)
        
        # 步骤4: 返回映射字典，不直接修改输入
        mapping = self._build_mapping(clusters)
        
        # 如果需要写回数据库
        if write_back:
            self._write_clusters_to_db(clusters, mgr)
        
        return mapping
    
    def _build_mapping(self, clusters) -> Dict[str, str]:
        """构建实体名映射"""
        mapping = {}
        for cluster in clusters:
            for member in cluster.members:
                if member != cluster.center:
                    mapping[member] = cluster.center
        return mapping
```

# pipeline/speaker_matcher.py
class SpeakerMatcher:
    """说话人匹配器"""
    
    def match_by_semantic(self, dialogue_text: str, candidates: List[Character]) -> Optional[MatchResult]:
        """基于语义相似度匹配说话人"""
        if not candidates:
            return None
        
        best_score = -1
        best_char = None
        
        for char in candidates:
            score = self.semantic_ranker.rank_dialogue(
                dialogue_text,
                char.get_dialogue_history()
            )
            if score > best_score:
                best_score = score
                best_char = char
        
        if best_char and best_score >= self.l2_threshold:
            # 修复：添加下限保护
            confidence = max(0.6, 0.6 + (best_score - self.l2_threshold) * 0.5)
            return MatchResult(
                character_name=best_char.name,
                confidence=confidence,
                match_type="semantic"
            )
        return None

        entities = self._detect_profession_titles(entities, text)

        entities = self._detect_position_titles(entities, text)
        
        # 6. 组织后缀识别

        # 7. 家族后缀识别

def setup_logger(name: str = "novel_tts", level: int = logging.INFO) -> logging.Logger:
    """
    配置日志
    
    参数:
        name: 日志名称
        level: 日志级别
    """

    logger.setLevel(level)
    
    # 避免重复添加handler
    if logger.handlers:
        return logger
    
    # 格式化器
    formatter = logging.Formatter(
        '%(asctime)s - %(name)s - %(levelname)s - %(message)s',
        datefmt='%Y-%m-%d %H:%M:%S'
    )
    
    # 控制台handler
    console_handler = logging.StreamHandler()
    console_handler.setLevel(level)
    console_handler.setFormatter(formatter)
    logger.addHandler(console_handler)
    
    # 文件handler（带轮转）
    file_handler = RotatingFileHandler(
        LOG_FILE,
        maxBytes=10*1024*1024,  # 10MB
        backupCount=5,
        encoding='utf-8'
    )
    file_handler.setLevel(level)
    file_handler.setFormatter(formatter)
    logger.addHandler(file_handler)
    
    return logger

# pipeline/emotion_tagger.py
# -*- coding: utf-8 -*-
"""情绪标注模块"""
from typing import List, Dict
import re

class EmotionTagger:
    """情绪标注器"""
    
    # 情绪模式
    EMOTION_PATTERNS = {
        "joy": [r"笑", r"开心", r"哈哈", r"高兴", r"喜悦", r"欢"],
        "anger": [r"怒", r"愤怒", r"可恶", r"生气", r"恼", r"愤"],
        "sadness": [r"哭", r"悲伤", r"唉", r"叹气", r"落泪", r"伤心"],
        "surprise": [r"惊讶", r"竟然", r"什么", r"吃惊", r"没想到"],
        "fear": [r"害怕", r"危险", r"可怕", r"恐惧", r"惊恐"],
    }
    
    def tag(self, text: str) -> Dict[str, float]:
        """
        标注文本情绪（支持多标签）
        
        返回:
            {情绪类型: 置信度}
            例如: {"joy": 0.8, "surprise": 0.3}
        """
        emotions = {}
        
        for emotion, patterns in self.EMOTION_PATTERNS.items():
            score = 0.0
            for pattern in patterns:
                if re.search(pattern, text):
                    score += 0.2  # 每个匹配增加0.2
            
            if score > 0:
                emotions[emotion] = min(score, 1.0)
        
        # 如果没有匹配的情绪，返回neutral
        if not emotions:
            emotions["neutral"] = 1.0
        
        return emotions
    
    def get_primary_emotion(self, text: str) -> str:
        """获取主要情绪"""
        emotions = self.tag(text)
        return max(emotions, key=emotions.get)
```
