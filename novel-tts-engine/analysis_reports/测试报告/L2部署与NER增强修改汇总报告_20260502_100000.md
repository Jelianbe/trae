```python
        text=token,

        start=i,
```

#### 1.5 新增Rule 5：组织名后缀识别（第327-343行）
```python
# Rule 5: Organization names with suffixes (暗影会/异能者联盟)
# Only match if token is 2+ chars (avoid false positives like "我会", "来公司")
if i + 1 < len(tokens) and (i + 1) not in rule_covered_indices:
    next_token = tokens[i + 1]
    if next_token in ORG_SUFFIXES and len(token) >= 2:
        combined = token + next_token
        rule_entities.append(Entity(
            text=combined,
            type='ORG',
            start=i,
            end=i + 2,
            confidence=0.85
        ))
        rule_entity_texts.add(combined)
        rule_covered_indices.update([i, i + 1])
        i += 2
        continue
```

### 5. `analysis_reports/evaluate_generalization_v3.py` - v3评估脚本集成L2

**修改原因**:
- 需要将L2语义排序层集成到正式评估脚本
- 需要解除对chapter_id的硬依赖
- 需要正确的对话历史建立逻辑

**修改内容 - `evaluate_speaker_matcher()`函数**:

```python
def evaluate_speaker_matcher(text, gt_dialogue_speakers, char_manager):
    """使用真实 SpeakerMatcher 管道，集成L2语义排序"""
    from pipeline.semantic_ranker import get_semantic_ranker
    
    # 初始化L2语义排序器
    semantic_ranker = get_semantic_ranker(enable_l2=True)
    semantic_ranker.load_model()
    matcher = SpeakerMatcher(
        character_manager=char_manager,
        semantic_ranker=semantic_ranker,
        l2_threshold=0.55,
    )
    
    # 构建GT映射
    gt_map = {}
    for item in gt_dialogue_speakers:
        text_key = item["text"][:30]
        gt_map[text_key] = item["speaker"]
    
    def find_gt_speaker(line_text):
        for key, speaker in gt_map.items():
            if key in line_text:
                return speaker
        return None
    
    def is_dialogue_line(line):
        if not line:
            return False
        return '\u201c' in line or '"' in line or "'" in line or '\u201d' in line
    
    lines = text.split('\n')
    current_chapter = 0
    prev_speaker = None
    correct = 0
    total = 0
    
    # 单遍扫描：匹配后立即更新对话历史
    for line in lines:
        line = line.strip()
        if not line:
            continue
        
        # 检测章节标题
        if re.match(r'(?:第[一二三四五六七八九十\d]+[章节回卷]|#{1,6}\s*第[一二三四五六七八九十\d]+[章节回卷])', line):
            current_chapter += 1
            continue
        
        if not is_dialogue_line(line):
            continue
        
        gt_speaker = find_gt_speaker(line)
        if not gt_speaker:
            continue
        
        total += 1
        ctx = DialogueContext(
            text=line,
            chapter_id=current_chapter,
            prev_speaker=prev_speaker
        )
        result = matcher.match_speaker(ctx)
        matched_name = result.character.name if result else None
        
        # 匹配成功则更新对话历史
        if matched_name == gt_speaker:
            correct += 1
            char = char_manager.get_character_by_name(gt_speaker)
            if char:
                matcher.update_activity(char.id, char.name)
            prev_speaker = matched_name
        else:
            # 即使匹配失败，也记录GT说话人以维持对话连续性
            char = char_manager.get_character_by_name(gt_speaker)
            if char:
                matcher.update_activity(char.id, char.name)
            prev_speaker = gt_speaker
    
    if total == 0:
        return 100.0
    return round(correct / total * 100, 1)
```

**关键改进**:
- 集成`SemanticRanker`到`evaluate_speaker_matcher()`
- 单遍扫描+实时对话历史更新
- 按章节分段处理，自动检测章节标题
- 匹配成功/失败都更新对话历史以维持连续性

```python
# -*- coding: utf-8 -*-
"""
L2语义排序层对比测试 v2
先构建对话历史，再评估L2效果。
"""
import sys
from pathlib import Path
import os
import re

sys.path.insert(0, str(Path(__file__).parent.parent))
os.environ['DEBUG_NER'] = '0'

import pipeline.nlp_basics as nlp_basics
original_analyze = nlp_basics.NLPBasics.analyze
def quiet_analyze(self, text):
    import sys
    old_stdout = sys.stdout
    sys.stdout = open(os.devnull, 'w')
    try:
        result = original_analyze(self, text)
    finally:
        sys.stdout = old_stdout
    return result
nlp_basics.NLPBasics.analyze = quiet_analyze

from pipeline.speaker_matcher import SpeakerMatcher, DialogueContext
from pipeline.character_manager import CharacterManager
from pipeline.semantic_ranker import get_semantic_ranker
import json

def evaluate_with_l2(novel_text, gt_dialogue_speakers, char_manager, enable_l2=False):
    """评估说话人匹配，可选启用L2。"""
    if enable_l2:
        semantic_ranker = get_semantic_ranker(enable_l2=True)
        semantic_ranker.load_model()
        matcher = SpeakerMatcher(character_manager=char_manager, semantic_ranker=semantic_ranker, l2_threshold=0.55)
    else:
        matcher = SpeakerMatcher(character_manager=char_manager, semantic_ranker=None)
    
    # 评估 - 按章节顺序处理
    correct = 0
    total = 0
    prev_speaker = None
    current_chapter = 0
    
    gt_map = {}
    for item in gt_dialogue_speakers:
        text_key = item["text"][:30]
        gt_map[text_key] = item["speaker"]
    
    # 按章节分段处理
    lines = novel_text.split('\n')
    chapter_lines = []
    for line in lines:
        if re.match(r'(?:第[一二三四五六七八九十\d]+[章节回卷]|#{1,6}\s*第[一二三四五六七八九十\d]+[章节回卷])', line):
            if chapter_lines:
                # 处理当前章节
                for line_text in chapter_lines:
                    line_text = line_text.strip()
                    if not line_text or '"' not in line_text:
                        continue
                    
                    gt_speaker = None
                    for key, speaker in gt_map.items():
                        if key in line_text:
                            gt_speaker = speaker
                            break
                    if not gt_speaker:
                        continue
                    
                    total += 1
                    ctx = DialogueContext(
                        text=line_text,
                        chapter_id=current_chapter,
                        prev_speaker=prev_speaker
                    )
                    result = matcher.match_speaker(ctx)
                    matched_name = result.character.name if result else None
                    
                    if matched_name == gt_speaker:
                        correct += 1
                        prev_speaker = matched_name
                    else:
                        prev_speaker = gt_speaker
                
                current_chapter += 1
                chapter_lines = []
        
        chapter_lines.append(line)
    
    # 处理最后一章
    for line_text in chapter_lines:
        line_text = line_text.strip()
        if not line_text or '"' not in line_text:
            continue
        
        gt_speaker = None
        for key, speaker in gt_map.items():
            if key in line_text:
                gt_speaker = speaker
                break
        if not gt_speaker:
            continue
        
        total += 1
        ctx = DialogueContext(
            text=line_text,
            chapter_id=current_chapter,
            prev_speaker=prev_speaker
        )
        result = matcher.match_speaker(ctx)
        matched_name = result.character.name if result else None
        
        if matched_name == gt_speaker:
            correct += 1
            prev_speaker = matched_name
        else:
            prev_speaker = gt_speaker
    
    if total == 0:
        return 100.0
    return round(correct / total * 100, 1)
```

```python
# ============================================================
# NLP 基类
# ============================================================

class NLPBasics:
    """
    NLP 基础处理模块
    - 分词
    - 词性标注
    - 命名实体识别(NER)
    - 规则增强 NER
    """
    def __init__(self):
        self.hanlp = None
        if HANLP_AVAILABLE:
            try:
                self.hanlp = HanLP.newSegment()
                print("✅ HanLP 加载成功")
            except Exception as e:
                print(f"❌ HanLP 加载失败: {e}")
    
    def segment(self, text: str) -> List[Token]:
        """
        分词
        """
        if not text:
            return []
        
        if not self.hanlp:
            # fallback: 按字切分
            return [Token(c, 'X', i, i + 1) for i, c in enumerate(text)]
        
        terms = self.hanlp.seg(text)
        tokens = []
        start = 0
        for term in terms:
            word = str(term.word)
            length = len(word)
            pos = str(term.nature)
            tokens.append(Token(
                text=word,
                pos=pos,
                start=start,
                end=start + length
            ))
            start += length
        
        return tokens
    
    def extract_entities(self, text: str) -> List[Entity]:
        """
        命名实体识别(NER)
        """
        if not text or not self.hanlp:
            return []
        
        # 使用 HanLP 的 NER
        recognized_entities = []
        try:
            # 尝试使用 HanLP 的 NER
            analyzer = HanLP.newSegment().enableCustomDictionaryForcing(True).enablePlaceRecognize(True).enablePersonRecognition(True).enableOrganizationRecognition(True)
            terms = analyzer.seg(text)
            
            current_entity = None
            for term in terms:
                word = str(term.word)
                pos = str(term.nature)
                
                # HanLP 的实体标签
                if pos in ('nr', 'nrf'):  # 人名
                    if current_entity and current_entity.type == 'PER':
                        current_entity.text += word
                        current_entity.end += len(word)
                    else:
                        if current_entity:
                            recognized_entities.append(current_entity)
                        current_entity = Entity(text=word, type='PER', start=0, end=len(word))
                elif pos in ('ns', 'nt'):  # 地名
                    if current_entity and current_entity.type == 'LOC':
                        current_entity.text += word
                        current_entity.end += len(word)
                    else:
                        if current_entity:
                            recognized_entities.append(current_entity)
                        current_entity = Entity(text=word, type='LOC', start=0, end=len(word))
                elif pos == 'ni':  # 机构名
                    if current_entity and current_entity.type == 'ORG':
                        current_entity.text += word
                        current_entity.end += len(word)
                    else:
                        if current_entity:
                            recognized_entities.append(current_entity)
                        current_entity = Entity(text=word, type='ORG', start=0, end=len(word))
                else:
                    if current_entity:
                        recognized_entities.append(current_entity)
                        current_entity = None
            
            if current_entity:
                recognized_entities.append(current_entity)
                
        except Exception as e:
            if os.environ.get('DEBUG_NER') == '1':
                print(f"❌ NER 失败: {e}")
        
        # 应用规则增强
        return self._enhance_entities(text, recognized_entities)
    
    def _enhance_entities(self, text: str, hanlp_entities: List[Entity]) -> List[Entity]:
        """
        应用规则增强 NER
        """
        tokens = self.segment(text)
        token_texts = [t.text for t in tokens]
        pos_tags = [t.pos for t in tokens]
        
        # 检查 HanLP 已识别的实体
        hanlp_entity_texts = set(e.text for e in hanlp_entities)
        
        # 应用规则
        rule_entities = []
        rule_entity_texts = set()
        rule_covered_indices = set()
        
        # 找出被 HanLP 实体覆盖的 token 索引
        for entity in hanlp_entities:
            for i, token in enumerate(tokens):
                if entity.start <= token.start < entity.end:
                    rule_covered_indices.add(i)
        
        i = 0
        while i < len(tokens):
            if i in rule_covered_indices:
                i += 1
                continue
            
            token = tokens[i]
            pos = pos_tags[i] if i < len(pos_tags) else 'X'
            
            # Rule 1: Prefix titles (老X/小X/大X) where X is a surname
            if token in PREFIX_TITLES and i + 1 < len(tokens):
                next_token = tokens[i + 1]
                next_pos = pos_tags[i + 1] if i + 1 < len(pos_tags) else 'X'
                
                if next_token in SINGLE_CHAR_SURNAMES and next_pos in ('NR', 'nr'):
                    combined = token + next_token
                    rule_entities.append(Entity(
                        text=combined,
                        type='PER',
                        start=i,
                        end=i + 2,
                        confidence=0.90
                    ))
                    rule_entity_texts.add(combined)
                    rule_covered_indices.update([i, i + 1])
                    i += 2
                    continue
            
            # Rule 4: Standalone professional titles (博士/教授/医生等)
            if token in PROFESSION_TITLES:
                rule_entities.append(Entity(
                    text=token,
                    type='PER',
                    start=i,
                    end=i + 1,
                    confidence=0.85
                ))
                rule_entity_texts.add(token)
                rule_covered_indices.add(i)
                i += 1
                continue
            
            # Rule 5: Organization names with suffixes (暗影会/异能者联盟)
            # Only match if token is 2+ chars (avoid false positives like "我会", "来公司")
            if i + 1 < len(tokens) and (i + 1) not in rule_covered_indices:
                next_token = tokens[i + 1]
                if next_token in ORG_SUFFIXES and len(token) >= 2:
                    combined = token + next_token
                    rule_entities.append(Entity(
                        text=combined,
                        type='ORG',
                        start=i,
                        end=i + 2,
                        confidence=0.85
                    ))
                    rule_entity_texts.add(combined)
                    rule_covered_indices.update([i, i + 1])
                    i += 2
                    continue
            
            # Rule 2: Single surname + title words (陈管家/林少爷)
            if pos not in ('NR', 'nr'):
                i += 1
                continue
            
            if len(token) > 1:
                i += 1
                continue
            
            if token not in SINGLE_CHAR_SURNAMES:
                i += 1
                continue
            
            if i + 1 >= len(tokens) or (i + 1) in rule_covered_indices:
                i += 1
                continue
            
            next_token = tokens[i + 1]
            combined = token + next_token
            
            if next_token in TITLE_WORDS:
                rule_entities.append(Entity(
                    text=combined,
                    type='PER',
                    start=i,
                    end=i + 2,
                    confidence=0.95
                ))
                rule_entity_texts.add(combined)
                rule_covered_indices.update([i, i + 1])
                i += 2
                continue
            
            # Rule 3: Position titles (张总/李哥/王姐)
            if next_token in POSITION_SUFFIXES:
                rule_entities.append(Entity(
                    text=combined,
                    type='PER',
                    start=i,
                    end=i + 2,
                    confidence=0.90
                ))
                rule_entity_texts.add(combined)
                rule_covered_indices.update([i, i + 1])
                i += 2
                continue
            
            if next_token in FAMILY_SUFFIXES:
                rule_entities.append(Entity(
                    text=combined,
                    type='ORG',
                    start=i,
                    end=i + 2,
                    confidence=0.90
                ))
                rule_entity_texts.add(combined)
                rule_covered_indices.update([i, i + 1])
                i += 2
                continue
            
            if next_token in LOCATION_SUFFIXES:
                rule_entities.append(Entity(
                    text=combined,
                    type='LOC',
                    start=i,
                    end=i + 2,
                    confidence=0.85
                ))
                rule_entity_texts.add(combined)
                rule_covered_indices.update([i, i + 1])
                i += 2
                continue
            
            i += 1
        
        # 合并 HanLP 和规则的实体
        all_entities = hanlp_entities.copy()
        
        # 添加规则识别的新实体
        for rule_entity in rule_entities:
            # 检查是否与 HanLP 实体冲突
            if rule_entity.text not in hanlp_entity_texts:
                all_entities.append(rule_entity)
        
        # 调试输出
        if os.environ.get('DEBUG_NER') == '1':
            print(f"\n=== NER 调试 ===")
            print(f"文本: {text[:80]}...")

            print(f"规则识别: {[(e.text, e.type) for e in rule_entities]}")
            print(f"最终结果: {[(e.text, e.type) for e in all_entities]}")
        
        return all_entities
    
    def analyze(self, text: str) -> Dict:
        """
        完整分析文本
        """
        tokens = self.segment(text)
```

```python
# -*- coding: utf-8 -*-
"""
L2 语义排序层
使用 BGE-small-zh 模型进行句子嵌入，计算语义相似度。
支持 ONNX 和 PyTorch 双后端。
"""
from typing import List, Tuple, Dict, Optional
import numpy as np
from dataclasses import dataclass
import os
from pathlib import Path
import threading

# Optional imports
try:
    import torch
    from transformers import AutoTokenizer, AutoModel
    TORCH_AVAILABLE = True
except ImportError:
    TORCH_AVAILABLE = False

try:
    import onnxruntime as ort
    from transformers import AutoTokenizer
    ONNX_AVAILABLE = True
except ImportError:
    ONNX_AVAILABLE = False


# Default model path (relative to project root)
MODEL_NAME = "BAAI/bge-small-zh-v1.5"
ONNX_MODEL_PATH = "models/bge-small-onnx"
PYTORCH_MODEL_PATH = "models/bge-small-zh-v1.5"


@dataclass
class SemanticProfile:
    """Character's semantic profile for L2 ranking."""
    character_id: int
    name: str
    dialogue_history: List[str]  # Recent dialogues
    profile_embedding: Optional[np.ndarray] = None  # Cached embedding


class SemanticCache:
    """Thread-safe cache for text embeddings."""
    
    def __init__(self, max_size: int = 1000):
        self._cache: Dict[str, np.ndarray] = {}
        self._lock = threading.Lock()
        self._max_size = max_size
    
    def get(self, text: str) -> Optional[np.ndarray]:
        with self._lock:
            return self._cache.get(text)
    
    def put(self, text: str, embedding: np.ndarray):
        with self._lock:
            if len(self._cache) >= self._max_size:
                # Remove oldest entry
                oldest_key = next(iter(self._cache))
                del self._cache[oldest_key]
            self._cache[text] = embedding
    
    def clear(self):
        with self._lock:
            self._cache.clear()


class SemanticRanker:
    """
    L2 Semantic Ranking Layer
    
    Uses BGE-small-zh to encode sentences and rank candidates by cosine similarity.
    Supports both ONNX and PyTorch backends (ONNX preferred for performance).
    """
    
    def __init__(self, model_name: str = MODEL_NAME, cache_size: int = 1000, enable_l2: bool = True):
        self.model_name = model_name
        self._enable_l2 = enable_l2
        self._backend = None  # 'onnx' or 'pytorch'
        self._onnx_session = None
        self._onnx_tokenizer = None
        self._tokenizer = None
        self._model = None
        self._cache = SemanticCache(max_size=cache_size)
    
    def load_model(self):
        """Load model, preferring ONNX for performance."""
        if not self._enable_l2:
            print("ℹ️ L2 semantic ranking disabled")
            return
        
        print("🔄 正在加载 L2 语义排序模型...")
        
        # Try ONNX first (faster)
        if ONNX_AVAILABLE:
            try:
                self._load_onnx()
                print("✅ L2 语义排序模型加载成功 (ONNX backend)")
                return
            except Exception as e:
                print(f"⚠️ ONNX backend 加载失败: {e}")
        
        # Fallback to PyTorch
        if TORCH_AVAILABLE:
            try:
                self._load_pytorch()
                print("✅ L2 语义排序模型加载成功 (PyTorch backend)")
                return
            except Exception as e:
                print(f"⚠️ PyTorch backend 加载失败: {e}")
        
        print("❌ L2 语义排序模型加载失败，将使用回退匹配策略")
        self._enable_l2 = False
    
    def _load_onnx(self):
        """Load ONNX model."""
        self._onnx_session = ort.InferenceSession(
            str(Path(ONNX_MODEL_PATH) / "model.onnx"),
            providers=['CPUExecutionProvider']
        )
        self._onnx_tokenizer = AutoTokenizer.from_pretrained(ONNX_MODEL_PATH)
        self._backend = 'onnx'
    
    def _load_pytorch(self):
        """Load PyTorch model."""
        self._tokenizer = AutoTokenizer.from_pretrained(PYTORCH_MODEL_PATH)
        self._model = AutoModel.from_pretrained(PYTORCH_MODEL_PATH)
        self._model.eval()
        self._backend = 'pytorch'
    
    def is_available(self) -> bool:
        """Check if L2 semantic ranking is available."""
        return self._enable_l2 and self._backend is not None
    
    def encode(self, text: str) -> Optional[np.ndarray]:
        """Encode a single text to embedding."""
        if not self.is_available():
            return None
        
        # Check cache first
        cached = self._cache.get(text)
        if cached is not None:
            return cached
        
        # Encode
        if self._backend == 'onnx':
            emb = self._encode_onnx(text)
        else:
            emb = self._encode_pytorch(text)
        
        # Cache the result
        self._cache.put(text, emb)
        return emb
    
    def encode_batch(self, texts: List[str]) -> Optional[np.ndarray]:
        """Encode a batch of texts to embeddings."""
        if not self.is_available() or not texts:
            return None
        
        # Check cache for all texts
        results = []
        uncached_texts = []
        uncached_indices = []
        
        for i, text in enumerate(texts):
            cached = self._cache.get(text)
            if cached is not None:
                results.append((i, cached))
            else:
                uncached_texts.append(text)
                uncached_indices.append(i)
        
        # If all cached, return
        if not uncached_texts:
            return np.array([r[1] for r in sorted(results, key=lambda x: x[0])])
        
        # Encode uncached texts
        if self._backend == 'onnx':
            embs = self._encode_batch_onnx(uncached_texts)
        else:
            embs = self._encode_batch_pytorch(uncached_texts)
        
        # Cache results
        for idx, emb in zip(uncached_indices, embs):
            self._cache.put(texts[idx], emb)
        
        # Merge cached and new embeddings
        final_embs = [None] * len(texts)
        for idx, emb in results:
            final_embs[idx] = emb
        for idx, emb in zip(uncached_indices, embs):
            final_embs[idx] = emb
        
        return np.array(final_embs)
    
    def _encode_onnx(self, text: str) -> np.ndarray:
        """Encode using ONNX model."""
        inputs = self._onnx_tokenizer(
            text,
            return_tensors="np",
            padding=True,
            truncation=True,
            max_length=256
        )
        
        outputs = self._onnx_session.run(None, {
            "input_ids": inputs["input_ids"],
            "attention_mask": inputs["attention_mask"],
        })
        
        # Mean pooling
        last_hidden_state = outputs[0]
        attention_mask = inputs["attention_mask"]
        mask_expanded = np.expand_dims(attention_mask, -1)
        sum_embeddings = np.sum(last_hidden_state * mask_expanded, axis=1)
        sum_mask = np.clip(mask_expanded.sum(axis=1), a_min=1e-9, a_max=None)
        embedding = sum_embeddings / sum_mask
        
        # Normalize
        norm = np.linalg.norm(embedding, axis=-1, keepdims=True)
        return (embedding / norm).squeeze(0)
    
    def _encode_pytorch(self, text: str) -> np.ndarray:
        """Encode using PyTorch model."""
        import torch
        
        inputs = self._tokenizer(
            text,
            return_tensors="pt",
            padding=True,
            truncation=True,
            max_length=256
        )
        
        with torch.no_grad():
            outputs = self._model(**inputs)
            last_hidden_state = outputs.last_hidden_state
            attention_mask = inputs["attention_mask"]
            mask_expanded = attention_mask.unsqueeze(-1).expand(last_hidden_state.size()).float()
            sum_embeddings = torch.sum(last_hidden_state * mask_expanded, 1)
            sum_mask = mask_expanded.sum(1).clamp(min=1e-9)
            embedding = sum_embeddings / sum_mask
        
        # Normalize
        embedding = embedding / torch.norm(embedding, dim=-1, keepdim=True)
        return embedding.squeeze(0).numpy()
    
    def _encode_batch_onnx(self, texts: List[str]) -> np.ndarray:
        """Encode a batch of texts using ONNX."""
        inputs = self._onnx_tokenizer(
            texts,
            return_tensors="np",
            padding=True,
            truncation=True,
            max_length=256
        )
        
        outputs = self._onnx_session.run(None, {
            "input_ids": inputs["input_ids"],
            "attention_mask": inputs["attention_mask"],
        })
        
        last_hidden_state = outputs[0]
        attention_mask = inputs["attention_mask"]
        mask_expanded = np.expand_dims(attention_mask, -1)
        sum_embeddings = np.sum(last_hidden_state * mask_expanded, axis=1)
        sum_mask = np.clip(mask_expanded.sum(axis=1), a_min=1e-9, a_max=None)
        embeddings = sum_embeddings / sum_mask
        
        # Normalize
        norm = np.linalg.norm(embeddings, axis=-1, keepdims=True)
        return (embeddings / norm).squeeze(1)
    
    def _encode_batch_pytorch(self, texts: List[str]) -> np.ndarray:
        """Encode a batch of texts using PyTorch."""
        import torch
        
        inputs = self._tokenizer(
            texts,
            return_tensors="pt",
            padding=True,
            truncation=True,
            max_length=256
        )
        
        with torch.no_grad():
            outputs = self._model(**inputs)
            last_hidden_state = outputs.last_hidden_state
            attention_mask = inputs["attention_mask"]
            mask_expanded = attention_mask.unsqueeze(-1).expand(last_hidden_state.size()).float()
            sum_embeddings = torch.sum(last_hidden_state * mask_expanded, 1)
            sum_mask = mask_expanded.sum(1).clamp(min=1e-9)
            embeddings = sum_embeddings / sum_mask
        
        # Normalize
        embeddings = embeddings / torch.norm(embeddings, dim=-1, keepdim=True)
        return embeddings.numpy()
    
    def rank(self, sentence: str, candidates: List[str], threshold: float = 0.7) -> List[Tuple[str, float]]:
        """
        Rank candidates by semantic similarity to the sentence.
        
        Args:
            sentence: The input sentence
            candidates: List of candidate names/descriptions
            threshold: Minimum similarity score to include
            
        Returns:
            List of (candidate, score) tuples, sorted by score descending
        """
        if not self.is_available() or not candidates:
            return []
        
        sentence_emb = self.encode(sentence)
        if sentence_emb is None:
            return []
        
        results = []
        for candidate in candidates:
            candidate_emb = self.encode(candidate)
            if candidate_emb is None:
                continue
            
            score = self.cosine_similarity(sentence_emb, candidate_emb)
            if score >= threshold:
                results.append((candidate, score))
        
        # Sort by score descending
        return sorted(results, key=lambda x: x[1], reverse=True)
    
    def rank_with_profiles(self, sentence: str, profiles: List[SemanticProfile], threshold: float = 0.55) -> List[Tuple[int, float]]:
        """
        Rank candidates using their semantic profiles.
        
        Args:
            sentence: The input sentence
            profiles: List of character semantic profiles
            threshold: Minimum similarity score
            
        Returns:
            List of (character_id, score) tuples, sorted by score descending
        """
        if not self.is_available() or not profiles:
            return []
        
        sentence_emb = self.encode(sentence)
        if sentence_emb is None:
            return []
        
        results = []
        for profile in profiles:
            if profile.profile_embedding is None:
                continue
            
            score = self.cosine_similarity(sentence_emb, profile.profile_embedding)
            if score >= threshold:
                results.append((profile.character_id, score))
        
        return sorted(results, key=lambda x: x[1], reverse=True)
    
    @staticmethod
    def cosine_similarity(emb1: np.ndarray, emb2: np.ndarray) -> float:
        """Calculate cosine similarity between two embeddings."""
        norm1 = np.linalg.norm(emb1)
        norm2 = np.linalg.norm(emb2)
        if norm1 == 0 or norm2 == 0:
            return 0.0
        return float(np.dot(emb1, emb2) / (norm1 * norm2))

    global _ranker_instance
    if _ranker_instance is None:
        _ranker_instance = SemanticRanker(model_name=model_name, enable_l2=enable_l2)
    return _ranker_instance
```
