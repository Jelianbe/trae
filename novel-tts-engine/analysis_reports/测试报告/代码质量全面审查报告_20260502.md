```python
    if _semantic_ranker is None:
        with _lock:
            if _semantic_ranker is None:  # 双重检查锁
                _semantic_ranker = SemanticRanker()
    return _semantic_ranker
```

**影响**: 并发场景下可能创建多个实例或崩溃。

#### P0-2: 上下文多样性验证存在性能瓶颈

**文件**: `pipeline/context_diversity_validator.py`
**问题说明**: 验证器需要对每个实体进行全量扫描，时间复杂度 O(n*m)，其中 n 是实体数，m 是文本长度。

**问题代码**:
```python
# pipeline/context_diversity_validator.py:L80-L120
    def validate_entity(self, entity: Entity, text: str) -> float:
        """验证实体的上下文多样性"""
        entity_text = entity.text
        positions = []
        start = 0
        while True:
            pos = text.find(entity_text, start)  # ❌ 每次都是字符串查找
            if pos == -1:
                break
            positions.append(pos)
            start = pos + 1
        
        # 计算右邻字种类
        right_chars = set()
        for pos in positions:
            end_pos = pos + len(entity_text)
            if end_pos < len(text):
                right_chars.add(text[end_pos])
        
        # ... 根据出现次数、右邻字种类调整置信度
```

**影响**: 大文本处理缓慢，尤其是当实体数量多时。

**改进建议**: 使用正则表达式一次匹配所有位置，或建立索引。

**改进建议**: 使用插件化架构：
```python
class PipelineStep(ABC):
    @abstractmethod
    def execute(self, context: PipelineContext) -> PipelineContext:
        pass

class PipelineRunner:
    def __init__(self):
        self.steps: List[PipelineStep] = []
    
    def add_step(self, step: PipelineStep):
        self.steps.append(step)
    
    def run(self, context: PipelineContext) -> PipelineContext:
        for step in self.steps:
            context = step.execute(context)
        return context
```

```python
# -*- coding: utf-8 -*-
"""实体链接模块 - 将NER识别的实体链接到角色库"""
from typing import List, Optional
from dataclasses import dataclass
from pipeline.character_manager import CharacterManager, get_character_manager


@dataclass
class LinkedEntity:
    """链接后的实体"""
    text: str
    type: str  # PER/LOC/ORG
    confidence: float
    standard_name: Optional[str] = None  # 标准名称
    is_linked: bool = False
    alias: Optional[str] = None
    start_pos: Optional[int] = None
    end_pos: Optional[int] = None


class EntityLinker:
    """实体链接器"""
    
    def __init__(self, char_manager=None):
        self.char_manager = char_manager or get_character_manager()
        self._link_cache = {}
    
    def link(self, entities: List, text: str, char_manager: CharacterManager = None) -> List[LinkedEntity]:
        """链接实体到角色库"""
        if char_manager:
            self.char_manager = char_manager  # ❌ 问题：直接修改实例属性
        
        linked_results = []
        for entity in entities:
            if entity.type != 'PER':
                # 非PER实体直接传递
                linked_results.append(LinkedEntity(
                    text=entity.text,
                    type=entity.type,
                    confidence=entity.confidence,
                    start_pos=entity.start,
                    end_pos=entity.end
                ))
                continue
            
            # 尝试链接到角色库
            linked_name = self._try_link(entity.text)
            if linked_name:
                # 链接成功，提升置信度
                linked_results.append(LinkedEntity(
                    text=entity.text,
                    type=entity.type,
                    confidence=max(entity.confidence, 0.8),
                    standard_name=linked_name,
                    is_linked=True,
                    start_pos=entity.start,
                    end_pos=entity.end
                ))
            else:
                # 未链接
                linked_results.append(LinkedEntity(
                    text=entity.text,
                    type=entity.type,
                    confidence=entity.confidence,
                    start_pos=entity.start,
                    end_pos=entity.end
                ))
        
        return linked_results
    
    def _try_link(self, entity_text: str) -> Optional[str]:
        """尝试将实体文本链接到角色库标准名"""
        # 直接匹配
        char = self.char_manager.get_character_by_name(entity_text)
        if char:
            return char.name
        
        # 别名匹配
        char = self.char_manager.get_character_by_alias(entity_text)
        if char:
            return char.name
        
        # 核心名提取（如"张总" → "张"）
        if len(entity_text) >= 2 and entity_text[-1] in '总哥姐叔伯':
            core_name = entity_text[:-1]
            char = self.char_manager.get_character_by_name(core_name)
            if char:
                return char.name
        
        return None


_entity_linker = None

def get_entity_linker(char_manager=None):
    """获取实体链接器单例"""
    global _entity_linker
    if _entity_linker is None:
        _entity_linker = EntityLinker(char_manager)
    return _entity_linker
```

```python
            chapter_id=chapter_id,
            title=chapter.title,
            sentences=sentence_data_list,
            statistics={
                "entity_count": len(linked_entities),

            }
        )
```
