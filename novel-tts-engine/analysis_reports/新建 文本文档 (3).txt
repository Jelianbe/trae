### 1. 在 `speaker_matcher.py` 中添加名称提取方法

```python
# 在 SpeakerMatcher 类中新增

import re
from typing import List, Set

def _extract_name_candidates_from_narration(self, text: str) -> List[str]:
    """
    从旁白文本中提取可能的角色名称候选（无预注册场景）
    返回：去重后的名称列表
    """
    if not text or len(text) < 2:
        return []
    
    candidates = set()
    
    # 方法1：HanLP NER 提取 PER 实体（需要 nlp_basics 实例）
    if hasattr(self, 'nlp') and self.nlp:
        try:
            ner_result = self.nlp.extract_entities(text)
            for ent in ner_result:
                if ent.type == 'PER' and len(ent.text) >= 2:
                    # 过滤明显非人名（保留长度2-10的中文或音译）
                    if re.search(r'[\u4e00-\u9fa5]{2,}', ent.text) or re.search(r'[A-Za-z]{3,}', ent.text):
                        candidates.add(ent.text.strip())
        except Exception as e:
            logger.debug(f"NER 提取失败: {e}")
    
    # 方法2：正则捕获“姓+职位”模式（如“赵总监”、“李经理”）
    # 支持常见姓氏 + 职位后缀
    surname = r'[赵钱孙李周吴郑王冯陈褚卫蒋沈韩杨朱秦尤许何吕施张孔曹严华金魏陶姜戚谢邹喻柏水窦章云苏潘葛奚范彭郎]'
    title = r'(?:总(?:监|裁|经理)|部长|局长|处长|科长|主任|队长|团长|法师|骑士|战士|导师|医师|教授|博士)'
    pattern_surname_title = re.compile(r'(' + surname + r'{1,2}' + title + r')')
    for match in pattern_surname_title.findall(text):
        candidates.add(match)
    
    # 方法3：正则捕获“名称+头衔”模式（如“亚瑟团长”、“艾琳法师”）
    # 允许中文名（2-4字）或音译名（2-6字母）后跟头衔
    name_part = r'(?:[\u4e00-\u9fa5]{2,4}|[A-Za-z]{2,6})'
    title_part = r'(?:团长|队长|法师|骑士|战士|导师|医师|教授|总监|经理|部长)'
    pattern_name_title = re.compile(r'(' + name_part + title_part + r')')
    for match in pattern_name_title.findall(text):
        candidates.add(match)
    
    # 方法4：普通中文姓名（2-3个汉字，非职位后缀）
    pattern_chinese_name = re.compile(r'([\u4e00-\u9fa5]{2,3})(?![总监经理部长团长队长])')
    for match in pattern_chinese_name.findall(text):
        # 避免匹配到常用词如“但是”、“因为”
        if match not in {'但是', '因为', '所以', '虽然', '然而', '于是', '这个', '那个'}:
            candidates.add(match)
    
    # 过滤明显非人名（物品、抽象词、方位词）
    blacklist = {'办公室', '会议室', '笔记本', '皮椅', '茶杯', '符文', '魔法卷轴', '眼睛', '手', '头', '声音', '光芒', '气息', '数据', '压力', '能量', '上', '里', '边', '那边', '旁边', '[对话]', '[音效]', '[强调]'}
    filtered = [c for c in candidates if c not in blacklist and len(c) >= 2]
    
    # 去重并返回
    return list(set(filtered))
```
