### 变更 3 — 新增 `_is_word_boundary_match` 方法

```python
# _is_word_boundary_match
#
# 用途：判断 target 是否在 text 中以独立词形式出现（词边界匹配）
# 来源：中文文本处理——中文词边界定义为 target 前后字符为非中文字符或文本边界
# 边界：
#   - 前字符为非中文或文本开头 → 通过
#   - 后字符为非中文或文本结尾 → 通过
#   - 否则拒绝（target 是其他词的一部分）
# 上限：12 行代码，不引入外部依赖
# 更新日期：2026-05-16
# 维护者：v7.0 别名匹配 Bug 修复
@staticmethod
def _is_word_boundary_match(text: str, target: str) -> bool:
    """判断 target 是否在 text 中以独立词形式出现（词边界匹配）。

    中文词边界定义：target 前后字符为非中文字符或文本边界。
    """
    idx = text.find(target)
    if idx < 0:
        return False
    # 前边界检查：开头或前一个字符非中文
    if idx > 0:
        prev_char = text[idx - 1]
        if '\u4e00' <= prev_char <= '\u9fff':
            return False
    # 后边界检查：结尾或后一个字符非中文
    end = idx + len(target)
    if end < len(text):
        next_char = text[end]
        if '\u4e00' <= next_char <= '\u9fff':
            return False
    return True
```

```python
def extract_srl_arg0s(self, text: str) -> List[str]:
    """从文本中提取 SRL ARG0（动作执行者）。

    用途：为 speaker_matcher 提供"谁做了动作"的句法级信号源
    来源：HanLP 多任务模型 SRL 输出，论元角色 'ARG0' 表示动作执行者
    边界：
      - 只返回 ARG0 文本，不返回索引
      - 多个谓词共享同一个 ARG0 时会去重
      - SRL 不可用/异常时返回空列表
    上限：每句 1-3 个 ARG0，无索引膨胀风险
    更新日期：2026-05-16
    维护者：v7.0 SRL ARG0 提取方案
    """
    if not self._initialized or not text or not text.strip():
        return []

    results = []
    try:
        doc = self.pipeline(text)
        srl_data = doc.get('srl', [])
    except Exception:
        return []

    seen = set()
    for pred_group in srl_data:
        for item in pred_group:
            if item[1] == 'ARG0':
                arg0_text = item[0]
                if arg0_text not in seen:
                    seen.add(arg0_text)
                    results.append(arg0_text)

    return results
```

```python
    """
    # 步骤A1: 去除"的"前修饰
    parts = arg0_text.split('的')
    if len(parts) >= 2:
        candidate = parts[-1].strip()
        if len(candidate) >= 2:
            arg0_text = candidate

    if len(arg0_text) < 2:
        return None

    # 步骤A2: 去除方位介词后缀
    # 来源：中文方位介词封闭集
    for suffix in ['那边', '这边', '上面', '下面', '里面', '外面',
                   '上', '里', '边', '中', '外', '旁边']:
        if arg0_text.endswith(suffix) and len(arg0_text) > len(suffix) + 1:
            arg0_text = arg0_text[:-len(suffix)]

    if len(arg0_text) < 2:
        return None

    # 步骤B: 去头衔后缀（TITLE_WORDS 已在 nlp_basics.py 中有完整注释）
    for title in TITLE_WORDS:
        if arg0_text.endswith(title) and len(arg0_text) > len(title) + 1:
            core = arg0_text[:-len(title)]
            cn_chars = [c for c in core if '\u4e00' <= c <= '\u9fff']
            if 1 <= len(cn_chars) <= 4:
                arg0_text = core
                break

    if len(arg0_text) < 2:
        return None

    # 步骤C: CHARACTER_BLACKLIST 非人物过滤（在 pattern_extractor.py 中有完整注释）
    if arg0_text in CHARACTER_BLACKLIST:
        return None
    for exclude in CHARACTER_BLACKLIST:
        if exclude in arg0_text:
            return None

    return arg0_text
```

```python
# _extract_srl_arg0_candidates
#
# 用途：从文本中提取 SRL ARG0 并计算距离衰减置信度
# 来源：v7.0 SRL ARG0 提取方案设计——距离为 0 时置信度最高（0.90）
# 边界：
#   - distance=0（同句主语）→ 0.90
#   - distance=1（邻句主语）→ 0.80
#   - distance>1（远句主语）→ 0.65
# 更新日期：2026-05-16
# 维护者：v7.0 SRL ARG0 提取方案
def _extract_srl_arg0_candidates(self, text: str, dialogue_sentence_index: int) -> List[Tuple[str, float]]:
    """从文本中提取 SRL ARG0 并计算距离衰减置信度。"""
    if not text or not text.strip():
        return []

    raw_arg0s = self.nlp.extract_srl_arg0s(text)
    if not raw_arg0s:
        return []

    candidates = []
    seen = set()
    for arg0_text in raw_arg0s:
        normalized = self._normalize_srl_arg0(arg0_text)
        if normalized and normalized not in seen:
            seen.add(normalized)
            distance = self._calc_srl_distance(text, arg0_text, dialogue_sentence_index)
            if distance == 0:
                confidence = 0.90
            elif distance == 1:
                confidence = 0.80
            else:
                confidence = 0.65
            candidates.append((normalized, confidence))

    return candidates


def _calc_srl_distance(self, text: str, arg0_text: str, dialogue_sentence_index: int) -> int:
    """计算 ARG0 所在句到对话句的距离。

    简化实现：以句号/问号/感叹号/省略号分句，计算 arg0_text 的
    首次出现句索引与 dialogue_sentence_index 的绝对差。
    """
    import re
    sentences = re.split(r'[。！？!?…]+', text)
    arg0_sentence_index = -1
    for i, sent in enumerate(sentences):
        if arg0_text in sent:
            arg0_sentence_index = i
            break
    if arg0_sentence_index < 0:
        return 99
    return abs(arg0_sentence_index - dialogue_sentence_index)
```

```python
# ───── 临时角色 ─────

def add_temp_character(self, name: str, project_id: int,
                       gender: str = 'unknown') -> bool:
    """注册临时角色（不入 SQLite）。

    跨章节继承：如果 _frequency 中已有此角色名
    （上一个章节出现过但未达标），作为初始计数。
    """
    if name in self._temp_chars:
        return False

    # 检查跨章节频次历史（_frequency 随 cleanup 不清空）
    prev_count = self._frequency.get((name, project_id), 0)

    self._temp_chars[name] = TempCharacterInfo(
        name=name, project_id=project_id, gender=gender,
        mention_count=prev_count,  # 继承跨章节累计
        first_seen_at=datetime.now()
    )

    # 如果历史频次已达阈值，直接晋升
    if prev_count >= PROMOTION_THRESHOLD:
        self._promote_temp_character(name, project_id)

    return True

def get_temp_character(self, name: str) -> Optional[TempCharacterInfo]:
    return self._temp_chars.get(name)

def get_all_temp_characters(self) -> List[TempCharacterInfo]:
    return list(self._temp_chars.values())

# ───── 频次 ─────

def increment_frequency(self, name: str, project_id: int):
    """角色名频次 +1。同时检测临时角色是否达到晋升阈值。

    _frequency 字典跨章节存活，不随 cleanup 清空。
    """
    key = (name, project_id)
    self._frequency[key] = self._frequency.get(key, 0) + 1

    # 检测临时角色晋升
    if name in self._temp_chars:
        self._temp_chars[name].mention_count += 1
        if self._temp_chars[name].mention_count >= PROMOTION_THRESHOLD:
            self._promote_temp_character(name, project_id)

def get_frequency(self, name: str, project_id: int) -> int:
    return self._frequency.get((name, project_id), 0)

# ───── 晋升 ─────

def _promote_temp_character(self, name: str, project_id: int):
    """将临时角色晋升为正式角色（写入 SQLite）。"""
    temp_info = self._temp_chars.get(name)
    if not temp_info:
        return None

    char = self.add_character(
        name=name,
        project_id=project_id,
        aliases=set(),
        gender=temp_info.gender
    )
    if char:
        logger.info(f"临时角色晋升: {name} (提及{temp_info.mention_count}次)")
        del self._temp_chars[name]
    return char

# ───── 清理 ─────

def cleanup_temp_characters(self):
    """章节结束时清理不达标的临时角色。

    不设 promote_all 参数——避免"章节末尾大赦"绕过阈值。
    未达标角色记录日志后清除；_frequency 保留，跨章节累计。
    """
    below = [
        info for info in self._temp_chars.values()
        if info.mention_count < PROMOTION_THRESHOLD
    ]
    if below:
        logger.debug(f"清理未达标临时角色: "
                     f"{[(t.name, t.mention_count) for t in below]}")
    self._temp_chars.clear()  # 清空临时角色，_frequency 保留
```

```
  # 用途：[这个词表用来做什么]

  # 边界：
  #   - [哪些东西应该往里加]
  #   - [哪些东西不应该往里加]
  # 上限：[约N条/词，不应无限制扩容]
  # 更新日期：[YYYY-MM-DD]
  # 维护者：[维护此词表的规则/责任人]
  
□ 如果是阈值，是否已外置到 config.py？
  必须支持 os.environ.get() 环境变量覆盖。
  
□ 是否触发了"需审批类"新增？
  静态词表（如新的黑名单集合）需走审批流程。
```
