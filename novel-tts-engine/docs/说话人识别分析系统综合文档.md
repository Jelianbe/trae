def _extract_narration(self, text, context_before='', context_after='') -> str:
    """从文本中提取旁白内容，去除引号内的对话"""
    narration_parts = []
    if context_before:
        narration_parts.append(re.sub(r'[「"『].*?[」"』]', '', context_before))
    if text:
        narration_parts.append(re.sub(r'[「"『].*?[」"』]', '', text))
    if context_after:
        narration_parts.append(re.sub(r'[「"『].*?[」"』]', '', context_after))
    narration = ' '.join(narration_parts)
    return narration.strip()
```

_LEFT_QUOTES = '\u0022\u300c\u300e'   # "「『
_RIGHT_QUOTES = '\u0022\u300d\u300f'  # "」』

def _build_quote_patterns():
    patterns = []
    for lq, rq in zip(_LEFT_QUOTES, _RIGHT_QUOTES):
        patterns.append(re.compile(
            rf'(?:^|[\n。！？；：,，\s])'
            rf'([^{re.escape(lq)}]*?)'       # group(1): 引号前内容
            rf'{re.escape(lq)}'
            rf'([^ {re.escape(rq)}]+?)'      # group(2): 对话内容
            rf'{re.escape(rq)}'
        ))
    return patterns  # DIALOGUE_PATTERNS (3个模式)
```

def resolve_in_local_window(self, gender, recent_speakers,
                            character_activity=None,
                            recent_mentions=None) -> Optional[MatchResult]:
    candidates = []
    
    # [增强] 从最近说话人中找性别匹配（优先级高）
    if recent_speakers:
        window_size = 10
        window_speakers = recent_speakers[-window_size:]
        for idx, speaker_name in enumerate(reversed(window_speakers)):
            char = self.char_manager.get_character_by_name(speaker_name)
            if char and char.gender == gender:
                position_score = 1.0 - (idx / len(window_speakers))
                activity_score = 0.0
                if character_activity:
                    activity_count = character_activity.get(char.id, 0)
                    activity_score = min(activity_count * 0.05, 0.2)
                combined_score = position_score + activity_score
                candidates.append((char, combined_score))
    
    # [新增] 从最近被提及角色中找性别匹配（补充信号）
    if recent_mentions:
        for idx, mentioned_name in enumerate(reversed(recent_mentions[-10:])):
            char = self.char_manager.get_character_by_name(mentioned_name)
            if char and char.gender == gender:
                # 避免与说话人候选重叠
                if not any(c.name == char.name for c, _ in candidates):
                    position_score = 1.0 - (idx / min(len(recent_mentions), 10))
                    activity_score = 0.0
                    if character_activity:
                        activity_count = character_activity.get(char.id, 0)
                        activity_score = min(activity_count * 0.03, 0.1)
                    combined_score = position_score + activity_score - 0.1
                    candidates.append((char, combined_score))
    
    if not candidates:
        return None
    
    best_char, best_score = max(candidates, key=lambda x: x[1])
    confidence = min(0.85 + best_score * 0.1, 0.95)
    return MatchResult(character=best_char, confidence=confidence,
                       match_type='pronoun_local_window')

def _extract_context_speakers(self, text, context_before, context_after):
    candidates = []

            if char:  # 角色库中找到
                candidates.append((char.name, '...', 0.95))
                continue
        if not char:
            char = self._register_temporary_character(name, narration)
        if char:
            candidates.append((char.name, '...', 0.95))
    
    # 2. 锁定角色旁白匹配（旁白中提及锁定角色）
    locked_char = self._match_from_character_library(narration, locked_only=True)
    if locked_char:
        candidates.append((locked_char.name, '锁定角色旁白匹配', 0.90))
    
    # 3. 普通角色库匹配
    normal_char = self._match_from_character_library(narration, locked_only=False)
    if normal_char:
        candidates.append((normal_char.name, '角色库旁白匹配', 0.85))
    
    # 4. 描述性角色提取
    descriptive_roles = self.role_extractor.extract(narration)
    for role in descriptive_roles:
        char = self.char_manager.get_character_by_name(role, ...)
        if not char:
            char = self._register_temporary_character(role, narration)
        if char:
            candidates.append((char.name, '描述性角色', 0.75))
    
    # 5. NER 兜底
    if not candidates:
        candidates.extend(self._extract_from_ner(narration))
    
    # 6. 最终兜底
    if not candidates:
        candidates.append(('未知', '无充分证据', 0.30))
    
    return candidates

  │     ├── _extract_narration()        → 去除引号内对话

  │     ├── _match_from_character_library()       → 角色库优先
  │     ├── role_extractor.extract()              → 描述性角色
  │     └── _extract_from_ner()                   → NER 兜底
  ├── prev_speaker_fallback（新增）

  │     └── pronoun_resolver.resolve_in_local_window()
