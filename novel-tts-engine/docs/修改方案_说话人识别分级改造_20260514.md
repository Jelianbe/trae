# 修改方案：说话人识别分级改造

> 日期：2026-05-14
> 来源：AI 聊天对话产出
> 目标：将人物识别从"勉强能用"（63%）推到"基本可靠"（预估 80-85%）

---

## 核心原则

1. **`SpeakerMatcher` 一行不改** —— 已经稳定运行、经历过 28 个 Bad Case 的打磨
2. **`Character.id` 就是 UID** —— 数据库自增主键已在，无需发明新标识符
3. **LLM 只在规则搞不定时才出场** —— 预估 20-30% 对话走 LLM，不可用时自动降级
4. **候选人质量决定一切** —— 用现有 HanLP NER + EntityCleaner 过滤过的角色库构建候选人

---

**生命周期**：纯内存，一章开始创建，一章结束清空

```python
class ChapterRoleCache:
    """章节级临时角色缓存。

    用途：
      1. 记录本章新发现的临时角色（避免污染项目库）
      2. 统计提及频率，为晋升提供依据
      3. 为 CandidateBuilder 提供"本章活跃角色"
    """

    def __init__(self):
        self._chars: Dict[str, Character] = {}      # name → Character(无 DB id)
        self._counts: Dict[str, int] = {}            # name → 提及次数

    def __len__(self) -> int:
        return len(self._chars)

    def add(self, name: str, gender: str = "unknown") -> Character:
        self._counts[name] = self._counts.get(name, 0) + 1
        if name not in self._chars:
            char = Character(name=name, gender=gender)
            self._chars[name] = char
        return self._chars[name]

    def get(self, name: str) -> Optional[Character]:
        return self._chars.get(name)

    def get_count(self, name: str) -> int:
        return self._counts.get(name, 0)

    def get_active(self, min_mentions: int = 1) -> List[Character]:
        return sorted(
            [c for c in self._chars.values() if self._counts.get(c.name, 0) >= min_mentions],
            key=lambda c: -self._counts.get(c.name, 0)
        )

    def get_promotable(self, min_mentions: int = 3) -> List[Character]:
        return self.get_active(min_mentions)

    def clear(self):
        self._chars.clear()
        self._counts.clear()
```

### 改动 1-3：`build_candidate_list()` 方法

**文件**：`pipeline/character_manager.py`
**位置**：`CharacterManager` 类内，`get_all_characters()` 之后（约 L305）
**类型**：新增方法

```python
def build_candidate_list(
    self,
    project_id: str,
    chapter_cache: Optional["ChapterRoleCache"] = None,
    max_candidates: int = 6,
) -> List[CandidateInfo]:
    """构建 LLM 候选人列表，按优先级排序。

    max_candidates 设为 6 而非更大：
    - Benchmark 已验证 3-6 人列表下 LLM 准确率 100%
    - 超过 6 人未做过 benchmark，属未验证假设

    优先级：
      1. 用户锁定的角色 (is_locked=1) —— 用户手动确认的角色
      2. 本章活跃角色 (chapter_cache 中提及 >= 2 次)
      3. 项目已注册角色 —— 按 last_seen 降序

    去重：同一 uid 不重复出现。
    """
    seen_uids: Set[int] = set()
    candidates: List[CandidateInfo] = []

    # 优先级 1：锁定角色
    for char in self._get_locked_characters(project_id):
        if char.id not in seen_uids:
            seen_uids.add(char.id)
            candidates.append(CandidateInfo(
                uid=char.id, name=char.name, gender=char.gender,
                source="locked"
            ))

    # 优先级 2：本章活跃角色
    if chapter_cache:
        for char in chapter_cache.get_active(min_mentions=2):
            candidates.append(CandidateInfo(
                uid=None, name=char.name, gender=char.gender,
                source="chapter_active",
                mention_count=chapter_cache.get_count(char.name)
            ))

    # 优先级 3：项目已注册角色
    for char in self.get_all_characters(project_id):
        if char.id and char.id not in seen_uids:
            seen_uids.add(char.id)
            candidates.append(CandidateInfo(
                uid=char.id, name=char.name, gender=char.gender,
                source="project"
            ))

    return candidates[:max_candidates]
```

```python
def _get_locked_characters(self, project_id: str) -> List[Character]:
    """获取用户锁定的角色。"""
    conn = self._get_connection()
    try:
        rows = conn.execute(
            "SELECT * FROM characters WHERE project_id=? AND is_locked=1",
            (project_id,)
        ).fetchall()
        return [self._row_to_character(r) for r in rows]
    finally:
        self._release_connection(conn)
```

## Phase 2：HybridSpeakerMatcher

### 关键设计决策：LLM 置信度校准

规则系统的 `confidence` 和 LLM 返回的 `confidence` **不在同一尺度**：
- 规则系统：基于精确匹配(1.0)、模糊匹配(0.5-0.85)、语义相似度(0.3-0.7)等多层级评分
- LLM：1.5B 模型没有元认知能力，不能"自己判断"confidence

**策略**：LLM 结果使用固定 confidence 值：
- LLM 明确选择了候选人 → `confidence = 0.85`（与规则系统 L1 确定性规则持平）
- LLM 回答 "unknown" → 返回 None（与规则系统行为一致）

这样比较逻辑才成立：
```
规则 confidence >= 0.85 → 直接用（LLM 不会比规则更好）
规则 confidence < 0.6   → LLM 兜底（0.85 > 规则，LLM 胜出）
规则 confidence 0.6-0.85 → LLM 二次确认，取 LLM（0.85 >= 规则）
```

### 改动 2-2：`match_speaker()` 核心逻辑

```python
def match_speaker(self, context: DialogueContext) -> Optional[MatchResult]:
    """L1+L2 → 如果失败或低置信度 → L3 LLM"""
    result = self.rule_matcher.match_speaker(context)

    if result is None:
        # 规则完全没辙
        return self._llm_fallback(context)

    if result.confidence < self.LLM_THRESHOLD:  # 0.6
        # 规则有答案但不自信 → LLM 二次确认
        llm_result = self._llm_fallback(context)
        if llm_result and llm_result.confidence >= result.confidence:
            self.logger.info(
                f"LLM 覆盖规则结果: '{result.character.name}'({result.confidence}) "
                f"→ '{llm_result.character.name}'({llm_result.confidence})"
            )
            return llm_result
        return result

    return result
```

### 改动 2-3：`_llm_fallback()`

```python
def _llm_fallback(self, context: DialogueContext) -> Optional[MatchResult]:
    if not self._ensure_llm_loaded():
        return None  # 降级：不阻塞

    candidates = self.char_manager.build_candidate_list(
        self.current_project_id, self.chapter_cache
    )
    if not candidates:
        return None

    prompt = self._build_llm_prompt(context, candidates)
    response = self._llm_infer(prompt)
    uid_or_name = self._parse_llm_response(response, candidates)

    if uid_or_name == "unknown":
        return None

    return self._resolve_to_match_result(uid_or_name)
```

### 改动 2-3-1：`_resolve_to_match_result()` —— 置信度校准

```python
def _resolve_to_match_result(self, uid_or_name) -> Optional[MatchResult]:
    """将 LLM 返回的 uid/name 转换为 MatchResult。

    置信度策略：
    - LLM 明确选择了候选人 → confidence = 0.85（与规则 L1 确定性规则持平）
    - 1.5B 模型没有元认知能力，不能自己判断 confidence
    """
    # 按 uid 或 name 从 CharacterManager 查找
    char = self.char_manager.get_by_uid(uid_or_name) or \
           self.char_manager.get_by_name(uid_or_name, self.current_project_id)

    if char is None:
        return None

    return MatchResult(character=char, confidence=0.85, match_type="llm")
```

### 改动 2-4：`_ensure_llm_loaded()`

```python
def _ensure_llm_loaded(self) -> bool:
    if self._llm_loaded:
        return True
    try:
        from transformers import AutoModelForCausalLM, AutoTokenizer
        import torch

        self._llm_tokenizer = AutoTokenizer.from_pretrained(
            self.llm_model_name, trust_remote_code=True
        )
        device = "cuda" if torch.cuda.is_available() else "cpu"
        self._llm_model = AutoModelForCausalLM.from_pretrained(
            self.llm_model_name,
            torch_dtype=torch.float16 if device == "cuda" else torch.float32,
            device_map="auto" if device == "cuda" else None,
            trust_remote_code=True,
        )
        self._llm_device = device
        self._llm_loaded = True
        return True
    except Exception as e:
        self.logger.warning(f"LLM 加载失败，降级为纯规则模式: {e}")
        return False
```
