## 核心原则回顾

根据项目规范要求，所有代码必须遵循以下原则：

1. **通则推理**: 优先使用句法、词性、标点等通用语言特征做判断，关键词只能是辅助信号，绝不能作为唯一依据。
2. **不建新映射表**: 禁止新增任何"A→B"形式的静态映射表。如果确实需要将自称词关联到角色，只能通过上下文推理，找不到就标记为未知。
3. **未知优于误判**: 当置信度不足或证据缺失时，必须诚实返回"未知"或低置信度结果，不允许强行匹配到最接近的候选。
4. **语言学/统计学依据**: 任何保留的静态集合必须有明确来源——单字姓氏库要标注是百家姓还是网文高频虚构姓氏，情绪词要标注是通用词还是基于某文本的统计结果，写清楚来源和边界。

```python
    """基于数量和句法检测群体说话人，替代硬编码映射表
    
    规则：
    1. 包含数字+人/位 → GROUP:N（如"三人"→GROUP:3）

    match = re.search(r'([一二三四五六七八九十\d]+)[人位]', text)
    if match:
        num = self._chinese_to_int(match.group(1))
        return f'GROUP:{num}'
    
    # 模式2: 群体指示词
    crowd_words = {'众', '所有', '大家', '全体', '人人'}
    if any(w in text for w in crowd_words):
        return 'GROUP:CROWD'
    
    # 模式3: 齐声/异口同声
    if '异口同声' in text or '齐声' in text:
```

```python
def _detect_group_speaker(self, text: str) -> Optional[str]:
    """基于数量和句法检测群体说话人
    
    规则：
    1. 数字+人/位 → GROUP:N
    2. 群体指示词 → GROUP:CROWD
    3. 齐声/异口同声 → GROUP:CROWD
    """
    import re
    
    # 模式1: 数字+人/位
    match = re.search(r'([一二三四五六七八九十\d]+)[人位]', text)
    if match:
        num_str = match.group(1)
        num = self._chinese_number_to_int(num_str)
        return f'GROUP:{num}'
    
    # 模式2: 群体指示词
    if any(w in text for w in ('众人', '大家', '所有', '全体')):
        return 'GROUP:CROWD'
    
    # 模式3: 齐声/异口同声
    if any(w in text for w in ('齐声', '异口同声')):
        return 'GROUP:CROWD'
    
    return None

def _chinese_number_to_int(self, num_str: str) -> int:
    """中文数字转整数"""
    mapping = {'一': 1, '二': 2, '三': 3, '四': 4, '五': 5,
               '六': 6, '七': 7, '八': 8, '九': 9, '十': 10}
```

```python
def _detect_western_name_by_structure(self, tokens: List[str], pos_tags: List[str]) -> List[Entity]:
    """基于句法结构检测西方名字
    
    规则：
    1. 包含"·"分隔符的NR词 → 西方名字
    2. 连续2+个NR词，中间有"·"或空格 → 西方名字
    3. 非中文姓氏的NR词 + 西方头衔 → 西方名字+头衔
    """
    entities = []
    
    # 模式1: 包含"·"的NR词
    text = ''.join(tokens)
    if '·' in text:
        # 提取包含"·"的NR词
        for i, (token, pos) in enumerate(zip(tokens, pos_tags)):
            if pos in ('NR', 'nr') and '·' in token:
                entities.append(Entity(
                    text=token, type='PER', start=i, end=i+1, confidence=0.95
                ))
    
    # 模式2: 连续NR词（名+姓）
    i = 0
    while i < len(tokens):
        if pos_tags[i] in ('NR', 'nr'):
            # 检查后续是否有"·"或另一个NR
            j = i + 1
            while j < len(tokens):
                if tokens[j] in ('·', '-', ' '):
                    j += 1
                    continue
                if pos_tags[j] in ('NR', 'nr'):
                    merged = ''.join(tokens[i:j+1])
                    entities.append(Entity(
                        text=merged, type='PER', start=i, end=j+1, confidence=0.90
                    ))
                    i = j + 1
                    break
                else:
                    break
            else:
                i += 1
        else:
            i += 1
    
    return entities
```

```python
    """基于HanLP词性标注检测说话上下文
    
    规则：
    1. 包含说话动词(v) + 引号/冒号 → 说话上下文
    2. 包含动作动词(v) + 引号/冒号 → 动作上下文
    3. 降级到关键词匹配(仅作为辅助信号)
    """
    # 优先使用HanLP词性标注

        for token in result.tokens:
            # 动词且后面紧跟引号/冒号
            if token.pos in ('v', 'vd', 'VV'):
                if self._followed_by_quote_or_colon(context, token):
                    return True
        return False
    except Exception:
        # HanLP不可用时降级到关键词匹配
        return self._has_speech_context_fallback(context)

def _followed_by_quote_or_colon(self, context: str, token: Token) -> bool:
    """检查token后是否紧跟引号或冒号"""
    token_end = context.find(token.text) + len(token.text)
    following = context[token_end:token_end+5]
    return any(c in following for c in ('「', '『', '"', ''', '：', ':'))

    """
```
